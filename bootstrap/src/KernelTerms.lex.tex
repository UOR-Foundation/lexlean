definition{
  name: apps,
  parameters: [fn : Term, args : (List Term)],
  result: Term,
  body: (match args | List.nil => fn | List.cons head rest => (apps (ctor Term.app fn head) rest)),
  executable: true,
  recursive_argument: args
}

definition{
  name: countCase,
  parameters: [index : Nat, branches : (List Branch)],
  result: Nat,
  body: (match branches
    | List.nil => 0
    | List.cons head rest => (if ((project head ctor) =? index) then ((countCase index rest) + 1) else (countCase index rest))),
  executable: true,
  recursive_argument: branches
}

definition{
  name: bindersOf,
  parameters: [index : Nat, branches : (List Branch)],
  result: Nat,
  body: (match branches
    | List.nil => 0
    | List.cons head rest => (if ((project head ctor) =? index) then (project head binders) else (bindersOf index rest))),
  executable: true,
  recursive_argument: branches
}

definition{
  name: proofOf,
  parameters: [index : Nat, branches : (List Branch)],
  result: Term,
  body: (match branches
    | List.nil => (ctor Term.bvar 0)
    | List.cons head rest => (if ((project head ctor) =? index) then (project head proof) else (proofOf index rest))),
  executable: true,
  recursive_argument: branches
}

definition{
  name: strayFrom,
  parameters: [count : Nat, branches : (List Branch)],
  result: (Option Nat),
  body: (match branches
    | List.nil => (ctor Option.none [Nat])
    | List.cons head rest => (if ((project head ctor) <? count) then (strayFrom count rest) else (ctor Option.some [Nat] (project head ctor)))),
  executable: true,
  recursive_argument: branches
}

definition{
  name: wanted,
  parameters: [shape : Ctor, induct : Bool],
  result: Nat,
  body: (if induct then ((project shape fields) + (project shape recursive)) else (project shape fields)),
  executable: true
}

definition{
  name: checkFrom,
  parameters: [ctors : (List Ctor), index : Nat, branches : (List Branch), induct : Bool],
  result: Check,
  body: (match ctors
    | List.nil => (match (strayFrom index branches)
      | Option.none => (ctor Check.pass)
      | Option.some stray => (ctor Check.fail (ctor Refusal.unknownCase stray)))
    | List.cons head rest => (if ((countCase index branches) =? 0) then (ctor Check.fail (ctor Refusal.missingCase index)) else (if ((countCase index branches) =? 1) then (if ((bindersOf index branches) =? (wanted head induct)) then (checkFrom rest (index + 1) branches induct) else (ctor Check.fail (ctor Refusal.binderCount index))) else (ctor Check.fail (ctor Refusal.repeatedCase index))))),
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]},
  recursive_argument: ctors
}

definition{
  name: minorsFrom,
  parameters: [ctors : (List Ctor), index : Nat, branches : (List Branch)],
  result: (List Term),
  body: (match ctors
    | List.nil => (nil Term)
    | List.cons head rest => (cons (proofOf index branches) (minorsFrom rest (index + 1) branches))),
  executable: true,
  recursive_argument: ctors
}

definition{
  name: casesTerm,
  parameters: [
    casesOn : Nat,
    params : (List Term),
    motive : Term,
    scrutinee : Term,
    ctors : (List Ctor),
    branches : (List Branch)
  ],
  result: Produced,
  body: (match (checkFrom ctors 0 branches false)
    | Check.pass => (ctor Produced.made (apps (ctor Term.app (ctor Term.app (apps (ctor Term.const casesOn) params) motive) scrutinee) (minorsFrom ctors 0 branches)))
    | Check.fail refusal => (ctor Produced.refused refusal)),
  executable: true
}

definition{
  name: refusalOf,
  parameters: [result : Produced],
  result: Check,
  body: (match result
    | Produced.made value => (ctor Check.pass)
    | Produced.refused refusal => (ctor Check.fail refusal))
}

definition{
  name: declare,
  parameters: [name : Nat, statement : Term, produced : Produced],
  result: Declared,
  body: (match produced
    | Produced.made value => (ctor Declared.declared (record Decl {name := name, statement := statement, value := value}))
    | Produced.refused refusal => (ctor Declared.rejected refusal)),
  executable: true
}

definition{
  name: declaredType,
  parameters: [declared : Declared],
  result: (Option Term),
  body: (match declared
    | Declared.declared decl => (ctor Option.some [Term] (project decl statement))
    | Declared.rejected refusal => (ctor Option.none [Term]))
}

definition{
  name: whenOk,
  parameters: [produced : Produced, statement : Term],
  result: (Option Term),
  body: (match produced
    | Produced.made value => (ctor Option.some [Term] statement)
    | Produced.refused refusal => (ctor Option.none [Term]))
}

theorem{
  name: declare_binds_statement,
  parameters: [name : Nat, statement : Term, produced : Produced],
  statement: ((declaredType (declare name statement produced)) = (whenOk produced statement)),
  proof: cases{
    branches: [
      {proof: reflexivity{}, binders: [value], constructor: made},
      {proof: reflexivity{}, binders: [refusal], constructor: refused}
    ],
    scrutinee: produced
  }
}

definition{
  name: fnOf,
  parameters: [term : Term],
  result: Term,
  body: (match term
    | Term.bvar index => (ctor Term.sort 0)
    | Term.sort level => (ctor Term.sort 0)
    | Term.const name => (ctor Term.sort 0)
    | Term.app fn arg => fn
    | Term.lam domain body => (ctor Term.sort 0)
    | Term.pi domain body => (ctor Term.sort 0)),
  executable: true
}

definition{
  name: argOf,
  parameters: [term : Term],
  result: Term,
  body: (match term
    | Term.bvar index => (ctor Term.sort 0)
    | Term.sort level => (ctor Term.sort 0)
    | Term.const name => (ctor Term.sort 0)
    | Term.app fn arg => arg
    | Term.lam domain body => (ctor Term.sort 0)

  executable: true
}

definition{
  name: constOf,
  parameters: [term : Term],
  result: Nat,
  body: (match term
    | Term.bvar index => 0
    | Term.sort level => 0
    | Term.const name => (name + 1)
    | Term.app fn arg => 0
    | Term.lam domain body => 0
    | Term.pi domain body => 0),
  executable: true
}

definition{
  name: equationOf,
  parameters: [goal : Term],
  result: Equation,
  body: (if ((constOf (fnOf (fnOf (fnOf goal)))) =? 1) then (ctor Equation.equation (argOf (fnOf (fnOf goal))) (argOf (fnOf goal))) else (if ((constOf (fnOf (fnOf goal))) =? 3) then (ctor Equation.iff (argOf (fnOf goal))) else (ctor Equation.other))),
  executable: true
}

definition{
  name: equationCheck,
  parameters: [goal : Term],
  result: Check,
  body: (match (equationOf goal)
    | Equation.equation ty lhs => (ctor Check.pass)
    | Equation.iff lhs => (ctor Check.pass)
    | Equation.other => (ctor Check.fail (ctor Refusal.notAnEquation)))
}

definition{
  name: reflexivityTerm,
  parameters: [goal : Term],
  result: Produced,
  body: (match (equationOf goal)
    | Equation.equation ty lhs => (ctor Produced.made (ctor Term.app (ctor Term.app (ctor Term.const 1) ty) lhs))
    | Equation.iff lhs => (ctor Produced.made (ctor Term.app (ctor Term.const 3) lhs))
    | Equation.other => (ctor Produced.refused (ctor Refusal.notAnEquation))),
  executable: true
}

definition{
  name: decideTerm,
  parameters: [goal : Term, instance : Term],
  result: Produced,
  body: (ctor Produced.made (ctor Term.app (ctor Term.app (ctor Term.app (ctor Term.const 4) goal) instance) (ctor Term.app (ctor Term.app (ctor Term.const 1) (ctor Term.const 5)) (ctor Term.const 6)))),
  executable: true
}

definition{
  name: inductionTerm,
  parameters: [
    recursor : Nat,
    params : (List Term),
    motive : Term,
    scrutinee : Term,
    generals : (List Term),
    ctors : (List Ctor),
    branches : (List Branch)
  ],
  result: Produced,
  body: (match (checkFrom ctors 0 branches true)
    | Check.pass => (ctor Produced.made (apps (apps (ctor Term.app (apps (ctor Term.const recursor) params) motive) (minorsFrom ctors 0 branches)) (cons scrutinee generals)))
    | Check.fail refusal => (ctor Produced.refused refusal)),
  executable: true
}

definition{
  name: memberCheck,
  parameters: [names : (List Nat), name : Nat],
  result: Check,
  body: (match names
    | List.nil => (ctor Check.fail (ctor Refusal.notProduced name))
    | List.cons head rest => (if (head =? name) then (ctor Check.pass) else (memberCheck rest name))),
  executable: true,
  production: {effects: [allocation], targets: [rust-std]},
  recursive_argument: names
}

definition{
  name: allMembers,
  parameters: [required : (List Nat), names : (List Nat)],
  result: Check,
  body: (match required
    | List.nil => (ctor Check.pass)
    | List.cons head rest => (match (memberCheck names head)
      | Check.pass => (allMembers rest names)
      | Check.fail refusal => (ctor Check.fail refusal))),
  executable: true,
  recursive_argument: required
}

definition{
  name: constants,
  parameters: [names : (List Nat)],
  result: (List Term),
  body: (match names
    | List.nil => (nil Term)
    | List.cons head rest => (cons (ctor Term.const head) (constants rest))),
  executable: true,
  recursive_argument: names
}

definition{
  name: applyTerm,
  parameters: [theorem : Nat, produced : (List Nat), args : (List Term)],
  result: Produced,
  body: (match (memberCheck produced theorem)
    | Check.pass => (ctor Produced.made (apps (ctor Term.const theorem) args))
    | Check.fail refusal => (ctor Produced.refused refusal)),
  executable: true
}

definition{
  name: lawCheck,
  parameters: [law : Nat, registered : (List Nat), premises : (List Nat), produced : (List Nat)],
  result: Check,
  body: (match (memberCheck registered law)
    | Check.pass => (allMembers premises produced)
    | Check.fail refusal => (ctor Check.fail (ctor Refusal.unregisteredLaw law))),
  executable: true,
  production: {effects: [allocation], targets: [rust-std]}
}

definition{
  name: lawTerm,
  parameters: [
    law : Nat,
    registered : (List Nat),
    args : (List Term),
    premises : (List Nat),
    produced : (List Nat)
  ],
  result: Produced,
  body: (match (lawCheck law registered premises produced)
    | Check.pass => (ctor Produced.made (apps (apps (ctor Term.const law) args) (constants premises)))
    | Check.fail refusal => (ctor Produced.refused refusal)),
  executable: true
}
