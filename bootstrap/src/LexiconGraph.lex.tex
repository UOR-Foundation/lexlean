definition{
  name: importDiag,
  parameters: [packages : (List GraphPackage), owner : Bytes, kept : (List GraphDiagnostic), import : (Bytes * Bytes)],
  result: (List GraphDiagnostic),
  body: (let shown : Bytes := (cat owner (cat (lit ": import ") (quoted (refText (first import) (second import))))); (match (indexOf packages (first import))
    | Option.some found => (match (prim index packages found : (Option GraphPackage))
      | Option.some target => (if (prim equal (project target version) (second import) : Bool) then kept else (cons (plainDiag (lit "LLR3001") (cat shown (lit " does not match the loaded version"))) kept))
      | Option.none => kept)
    | Option.none => (cons (plainDiag (lit "LLR3001") (cat shown (lit " is not available"))) kept))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{

  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: countIn,

  executable: true

definition{
  name: availableVersions,

  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: reverseEdges,
  parameters: [packages : (List GraphPackage), available : (List (Bytes * Bytes))],
  result: (List (Bytes * Bytes)),
  body: (prim list_fold (fun [available, packages] (kept : (List (Bytes * Bytes))) (package : GraphPackage) => (if ((countIn packages (project package id)) =? 1) then (prim list_fold (fun [available, package] (inner : (List (Bytes * Bytes))) (import : (Bytes * Bytes)) => (match (LexLeanStd.Map::mapGet available (first import))
    | Option.some version => (if (prim equal version (second import) : Bool) then (cons ((refText (first import) (second import)), (refText (project package id) (project package version))) inner) else inner)
    | Option.none => inner)) kept (project package imports) : (List (Bytes * Bytes))) else kept)) (nil (Bytes * Bytes)) packages : (List (Bytes * Bytes))),
  axioms: [Quot.sound, propext],
  executable: true
}

  name: reachStep,

  result: (Option (List Bytes)),
  body: (let grown : (List Bytes) := (prim list_fold (fun [reached] (kept : (List Bytes)) (edge : (Bytes * Bytes)) => (if (LexLeanStd.Map::setContains reached (first edge)) then (LexLeanStd.Order::insertSorted kept (second edge)) else kept)) reached edges : (List Bytes)); (if ((prim length grown : Nat) =? (prim length reached : Nat)) then (ctor Option.none [(List Bytes)]) else (ctor Option.some [(List Bytes)] grown))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: packageImporters,

  result: (List Bytes),
  body: (let edges : (List (Bytes * Bytes)) := (reverseEdges packages (availableVersions packages members)); (let start : (List Bytes) := (prim list_fold (fun (kept : (List Bytes)) (member : (Bytes * Bytes)) => (LexLeanStd.Order::insertSorted kept (refText (first member) (second member)))) (nil Bytes) members : (List Bytes)); (first (prim iterate_until (fun [edges] (reached : (List Bytes)) => (reachStep edges reached)) ((prim length edges : Nat) + 2) start : ((List Bytes) * Bool))))),
  axioms: [Quot.sound, propext],
  executable: true,

  result: (List Nat),

  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: cycleDiag,
  parameters: [packages : (List GraphPackage), stack : (List (Nat * Nat)), target : Nat],
  result: GraphDiagnostic,
  body: (let bottomUp : (List Nat) := (prim list_fold (fun (kept : (List Nat)) (frame : (Nat * Nat)) => (cons (first frame) kept)) (nil Nat) stack : (List Nat)); (let onCycle : (List Nat) := (dropUntil target bottomUp); (let named : (List GraphPackage) := (LexLeanStd.Order::reverseList [GraphPackage] (prim list_fold (fun [packages] (kept : (List GraphPackage)) (node : Nat) => (match (prim index packages node : (Option GraphPackage))
    | Option.some package => (cons package kept)
    | Option.none => kept)) (nil GraphPackage) onCycle : (List GraphPackage))); (let targetId : Bytes := (match (prim index packages target : (Option GraphPackage))
    | Option.some package => (project package id)
    | Option.none => 0x); (let ids : (List Bytes) := (LexLeanStd.Order::reverseList [Bytes] (cons targetId (prim list_fold (fun (kept : (List Bytes)) (package : GraphPackage) => (cons (project package id) kept)) (nil Bytes) named : (List Bytes)))); (let members : (List (Bytes * Bytes)) := (LexLeanStd.Order::reverseList [(Bytes * Bytes)] (prim list_fold (fun (kept : (List (Bytes * Bytes))) (package : GraphPackage) => (cons ((project package id), (project package version)) kept)) (nil (Bytes * Bytes)) named : (List (Bytes * Bytes)))); (record GraphDiagnostic {code := (lit "LLR3003"), message := (cat (lit "package import cycle through ") (quoted targetId)), cycle := (ctor Option.some [((List Bytes) * (List Bytes))] (ids, (packageImporters packages members)))}))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: finishNode,
  parameters: [packages : (List GraphPackage), maxImportDepth : Nat, walk : Walk, node : Nat, package : GraphPackage, rest : (List (Nat * Nat))],
  result: Walk,
  body: (let deepest : Nat := (deepestOf packages (project walk depths) package); (record Walk {states := (setNat (project walk states) node 2), depths := (setNat (project walk depths) node deepest), order := (cons node (project walk order)), stack := rest, root := (project walk root), diags := (if (maxImportDepth <=? deepest) then (cons (depthDiag maxImportDepth deepest (project package id)) (project walk diags)) else (project walk diags)), cycle := (project walk cycle)})),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: followImport,
  parameters: [packages : (List GraphPackage), walk : Walk, node : Nat, next : Nat, import : (Bytes * Bytes), rest : (List (Nat * Nat))],
  result: Walk,
  body: (let target : Nat := (indexOr packages (first import)); (let stack : (List (Nat * Nat)) := (cons (node, (next + 1)) rest); (let state : Nat := (natAt (project walk states) target); (if (state =? 1) then (record Walk {states := (project walk states), depths := (project walk depths), order := (project walk order), stack := stack, root := (project walk root), diags := (project walk diags), cycle := (ctor Option.some [GraphDiagnostic] (cycleDiag packages stack target))}) else (if (state =? 2) then (record Walk {states := (project walk states), depths := (project walk depths), order := (project walk order), stack := stack, root := (project walk root), diags := (project walk diags), cycle := (project walk cycle)}) else (record Walk {states := (setNat (project walk states) target 1), depths := (project walk depths), order := (project walk order), stack := (cons (target, 0) stack), root := (project walk root), diags := (project walk diags), cycle := (project walk cycle)})))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: startRoot,
  parameters: [walk : Walk],
  result: Walk,
  body: (let root : Nat := (project walk root); (if ((natAt (project walk states) root) =? 2) then (record Walk {states := (project walk states), depths := (project walk depths), order := (project walk order), stack := (project walk stack), root := (root + 1), diags := (project walk diags), cycle := (project walk cycle)}) else (record Walk {states := (setNat (project walk states) root 1), depths := (project walk depths), order := (project walk order), stack := (cons (root, 0) (nil (Nat * Nat))), root := (root + 1), diags := (project walk diags), cycle := (project walk cycle)}))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: walkStep,
  parameters: [packages : (List GraphPackage), count : Nat, maxImportDepth : Nat, walk : Walk],
  result: (Option Walk),
  body: (match (project walk cycle)
    | Option.some found => (ctor Option.none [Walk])
    | Option.none => (match (project walk stack)
      | List.nil => (if (count <=? (project walk root)) then (ctor Option.none [Walk]) else (ctor Option.some [Walk] (startRoot walk)))
      | List.cons top rest => (match (prim index packages (first top) : (Option GraphPackage))
        | Option.none => (ctor Option.none [Walk])
        | Option.some package => (match (prim index (project package imports) (second top) : (Option (Bytes * Bytes)))
          | Option.some import => (ctor Option.some [Walk] (followImport packages walk (first top) (second top) import rest))
          | Option.none => (ctor Option.some [Walk] (finishNode packages maxImportDepth walk (first top) package rest)))))),
  axioms: [Quot.sound, propext],
  executable: true
}

  body: (prim list_fold (fun [closures] (visible : (List Bytes)) (import : (Bytes * Bytes)) => (LexLeanStd.Map::setUnion (LexLeanStd.Order::insertSorted visible (first import)) (lookupClosure closures (first import)))) (LexLeanStd.Order::insertSorted (nil Bytes) (project package id)) (project package imports) : (List Bytes)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: importClosures,
  parameters: [packages : (List GraphPackage), order : (List Nat)],
  result: (List (Bytes * (List Bytes))),
  body: (LexLeanStd.Order::reverseList [(Bytes * (List Bytes))] (prim list_fold (fun [packages] (closures : (List (Bytes * (List Bytes)))) (node : Nat) => (match (prim index packages node : (Option GraphPackage))
    | Option.some package => (cons ((project package id), (closureOf closures package)) closures)
    | Option.none => closures)) (nil (Bytes * (List Bytes))) order : (List (Bytes * (List Bytes))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: walkGraph,
  parameters: [packages : (List GraphPackage), maxImportDepth : Nat],
  result: GraphOutcome,
  body: (let count : Nat := (countPackages packages); (let start : Walk := (record Walk {states := (zeros packages), depths := (zeros packages), order := (nil Nat), stack := (nil (Nat * Nat)), root := 0, diags := (nil GraphDiagnostic), cycle := (ctor Option.none [GraphDiagnostic])}); (let walked : Walk := (first (prim iterate_until (fun [count, maxImportDepth, packages] (walk : Walk) => (walkStep packages count maxImportDepth walk)) ((count + count) + ((countImports packages) + 2)) start : (Walk * Bool))); (let diags : (List GraphDiagnostic) := (LexLeanStd.Order::reverseList [GraphDiagnostic] (project walked diags)); (match (project walked cycle)
    | Option.some found => (ctor GraphOutcome.GraphRefused (LexLeanStd.Order::reverseList [GraphDiagnostic] (cons found (project walked diags))))
    | Option.none => (match diags
      | List.cons head tail => (ctor GraphOutcome.GraphRefused diags)
      | List.nil => (let order : (List Nat) := (LexLeanStd.Order::reverseList [Nat] (project walked order)); (ctor GraphOutcome.GraphAccepted order (importClosures packages order))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: packageGraph,
  parameters: [packages : (List GraphPackage), maxImportDepth : Nat],
  result: GraphOutcome,
  body: (match (duplicateDiags packages)
    | List.cons head tail => (ctor GraphOutcome.GraphRefused (cons head tail))
    | List.nil => (match (importDiags packages)
      | List.cons head tail => (ctor GraphOutcome.GraphRefused (cons head tail))
      | List.nil => (walkGraph packages maxImportDepth))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow], targets: [rust-std]}
}
