definition{
  name: extractForm,
  parameters: [row : (List (Bytes * TomlRead::Value))],
  result: (Option RawForm),

    | Option.none => (ctor Option.none [RawForm])
    | Option.some formId => (match (getStr row (lit "channel"))
      | Option.none => (ctor Option.none [RawForm])
      | Option.some channel => (match (getStr row (lit "surface"))
        | Option.none => (ctor Option.none [RawForm])
        | Option.some surface => (match (getBool row (lit "canonical_source"))
          | Option.none => (ctor Option.none [RawForm])

            | Option.none => (ctor Option.none [RawForm])
            | Option.some features => (ctor Option.some [RawForm] (record RawForm {id := formId, channel := channel, surface := surface, canonicalSource := cs, features := features}))))))) else (ctor Option.none [RawForm])),
  executable: true

definition{
  name: extractForms,
  parameters: [rows : (List (List (Bytes * TomlRead::Value)))],
  result: (Option (List RawForm)),
  body: (match (prim list_fold (fun (acc : (Option (List RawForm))) (row : (List (Bytes * TomlRead::Value))) => (match acc
      | Option.none => (ctor Option.none [(List RawForm)])
      | Option.some soFar => (match (extractForm row)
        | Option.none => (ctor Option.none [(List RawForm)])
        | Option.some rf => (ctor Option.some [(List RawForm)] (cons rf soFar))))) (ctor Option.some [(List RawForm)] (nil RawForm)) rows : (Option (List RawForm)))
    | Option.some revList => (ctor Option.some [(List RawForm)] (LexLeanStd.Order::reverseList [RawForm] revList))
    | Option.none => (ctor Option.none [(List RawForm)])),
  executable: true
}

definition{
  name: extractEliminator,
  parameters: [doc : (List TomlRead::Section)],
  result: (Option (Option RawEliminator)),
  body: (match (TomlRead::docSection doc (cons (lit "eliminator") (nil Bytes)) false)
    | Option.none => (ctor Option.some [(Option RawEliminator)] (ctor Option.none [RawEliminator]))
    | Option.some sec => (let row : (List (Bytes * TomlRead::Value)) := (firstRow sec); (if (hasOnlyKeys row (eliminatorAllowedKeys)) then (match (getStr row (lit "cases_lean_name"))
      | Option.none => (ctor Option.none [(Option RawEliminator)])
      | Option.some cln => (match (getStr row (lit "induction_lean_name"))
        | Option.none => (ctor Option.none [(Option RawEliminator)])
        | Option.some iln => (match (constructorRowsOrEmpty doc row)
          | Option.none => (ctor Option.none [(Option RawEliminator)])
          | Option.some constructorRows => (match (extractConstructors constructorRows)
            | Option.none => (ctor Option.none [(Option RawEliminator)])
            | Option.some constructors => (ctor Option.some [(Option RawEliminator)] (ctor Option.some [RawEliminator] (record RawEliminator {casesLeanName := cln, inductionLeanName := iln, constructors := constructors}))))))) else (ctor Option.none [(Option RawEliminator)])))),
  executable: true
}

definition{
  name: extractRaw,
  parameters: [doc : (List TomlRead::Section)],
  result: (Option RawEntry),
  body: (let top : (List (Bytes * TomlRead::Value)) := (topRow doc); (if (hasOnlyKeys top (rootAllowedKeys)) then (match (getStr top (lit "spec"))
    | Option.none => (ctor Option.none [RawEntry])
    | Option.some spec => (match (getStr top (lit "id"))
      | Option.none => (ctor Option.none [RawEntry])
      | Option.some entryId => (match (getStr top (lit "category"))
        | Option.none => (ctor Option.none [RawEntry])
        | Option.some category => (match (getOptStr top (lit "signature"))
          | Option.none => (ctor Option.none [RawEntry])
          | Option.some signature => (match (getOptArr top (lit "universes"))
            | Option.none => (ctor Option.none [RawEntry])
            | Option.some universes => (match (getNat top (lit "surface_arity"))
              | Option.none => (ctor Option.none [RawEntry])
              | Option.some surfaceArity => (match (getStr top (lit "frame"))
                | Option.none => (ctor Option.none [RawEntry])
                | Option.some frame => (match (getOptNat top (lit "precedence"))
                  | Option.none => (ctor Option.none [RawEntry])
                  | Option.some precedence => (match (getOptStr top (lit "associativity"))
                    | Option.none => (ctor Option.none [RawEntry])
                    | Option.some associativity => (match (TomlRead::docSection doc (cons (lit "denotation") (nil Bytes)) false)
                      | Option.none => (ctor Option.none [RawEntry])
                      | Option.some dsec => (let drow : (List (Bytes * TomlRead::Value)) := (firstRow dsec); (match (extractDenotationRow drow)
                        | Option.none => (ctor Option.none [RawEntry])
                        | Option.some dfields => (let dKind : Bytes := (project dfields kind); (let dConstructor : (Option Bytes) := (project dfields constructor); (let dModule : (Option Bytes) := (project dfields module); (let dName : (Option Bytes) := (project dfields name); (let dComponent : (Option Bytes) := (project dfields component); (let dValue : (Option Bytes) := (project dfields value); (match (formRowsOrEmpty doc top)
                          | Option.none => (ctor Option.none [RawEntry])
                          | Option.some formRows => (match (extractForms formRows)
                            | Option.none => (ctor Option.none [RawEntry])
                            | Option.some forms => (let hasRender : Bool := (match (TomlRead::docSection doc (cons (lit "render") (nil Bytes)) false)
                              | Option.some s => true
                              | Option.none => false); (let rrow : (List (Bytes * TomlRead::Value)) := (match (TomlRead::docSection doc (cons (lit "render") (nil Bytes)) false)
                              | Option.some s => (firstRow s)
                              | Option.none => (nil (Bytes * TomlRead::Value))); (match (getOptStr rrow (lit "math"))
                              | Option.none => (ctor Option.none [RawEntry])
                              | Option.some renderMath => (match (getOptStr rrow (lit "text"))
                                | Option.none => (ctor Option.none [RawEntry])
                                | Option.some renderText => (let hasCalculation : Bool := (match (TomlRead::docSection doc (cons (lit "calculation") (nil Bytes)) false)
                                  | Option.some s => true
                                  | Option.none => false); (let crow : (List (Bytes * TomlRead::Value)) := (match (TomlRead::docSection doc (cons (lit "calculation") (nil Bytes)) false)
                                  | Option.some s => (firstRow s)
                                  | Option.none => (nil (Bytes * TomlRead::Value))); (match (if hasCalculation then (getStr crow (lit "kind")) else (ctor Option.some [Bytes] 0x))
                                  | Option.none => (ctor Option.none [RawEntry])
